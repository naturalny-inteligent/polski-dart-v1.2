<div id="ekran-podsumowania" class="container">
    <h1>Koniec Sezonu (Grudzień)</h1>
    <h2>Podsumowanie wyników</h2>

<script>
    // Zmienne globalne gracza
    let gracz = {
        imie: "", pseudonim: "", wiek: 0,
        srednia: 0, dable: 0, psychika: 100, exp: 0, mecze: 0
    };

    function losujZPrzedzialu(min, max) {
        return Math.floor(Math.random() * (max - min + 1)) + min;
    }

    function rozpocznijGry() {
        let wiekInput = document.getElementById('wiek').value;
        let msg = document.getElementById('error-msg');
        
        if (!wiekInput) { msg.innerText = "Wpisz wiek. Czytanie ze zrozumieniem boli?"; return; }
        gracz.wiek = parseInt(wiekInput);
        
        if (gracz.wiek < 13) { msg.innerText = "Jesteś za młody na ostre lotki. Zablokowane."; return; }
        if (gracz.wiek > 80) { msg.innerText = "Dziadku, daj sobie spokój. Serce nie wytrzyma emocji."; return; }

        gracz.imie = document.getElementById('imie').value || "Brak Danych";
        gracz.pseudonim = document.getElementById('pseudonim').value || "No-Name";
        
        let poziom = document.getElementById('poziom').value;
        if (poziom === "amator") gracz.srednia = losujZPrzedzialu(25, 35);
        else if (poziom === "barowy") gracz.srednia = losujZPrzedzialu(45, 55);
        else gracz.srednia = losujZPrzedzialu(65, 75);
        
        gracz.dable = Math.max(5, Math.floor(gracz.srednia / 2.5) + losujZPrzedzialu(-5, 5));

        document.getElementById('ekran-startowy').style.display = 'none';
        przejdzDoRozgrywek();
    }

    function przejdzDoRozgrywek() {
        document.getElementById('ekran-podsumowania').style.display = 'none';
        document.getElementById('ekran-rozgrywek').style.display = 'block';
        aktualizujStatystyki();
    }

    function aktualizujStatystyki() {
        document.getElementById('stat-srednia').innerText = gracz.srednia.toFixed(1);
        document.getElementById('stat-dable').innerText = gracz.dable;
        document.getElementById('stat-psychika').innerText = gracz.psychika;
        document.getElementById('stat-exp').innerText = gracz.exp;
    }

    function symulujSezon() {
        let wybory = document.getElementsByName('liga');
        let liga = "boty";
        for (let i = 0; i < wybory.length; i++) {
            if (wybory[i].checked) liga = wybory[i].value;
        }

        let zmianaSredniej = 0;
        let zmianaDabli = 0;
        let zmianaPsychiki = 0;
        let dodaneMecze = 0;
        let dodanyExp = 0;
        let fazaTurnieju = "Brak";
        let etapy = ["Grupa", "1/16 Finału", "1/8 Finału", "Ćwierćfinał", "Półfinał", "Finał", "Wygrana w Turnieju"];

        // Logika rozgrywek
        if (liga === "boty") {
            zmianaSredniej = losujZPrzedzialu(0, 2);
            zmianaDabli = losujZPrzedzialu(0, 1);
            zmianaPsychiki = 5; 
            dodaneMecze = losujZPrzedzialu(30, 60);
            dodanyExp = 10;
            fazaTurnieju = "Grasz tylko w piwnicy";
        } 
        else if (liga === "dartcounter") {
            zmianaSredniej = losujZPrzedzialu(-1, 3);
            zmianaDabli = losujZPrzedzialu(-2, 3);
            zmianaPsychiki = losujZPrzedzialu(-10, 5);
            dodaneMecze = losujZPrzedzialu(20, 40);
            dodanyExp = 25;
            fazaTurnieju = etapy[losujZPrzedzialu(0, 2)]; 
        }
        else if (liga === "poznan") {
            zmianaSredniej = losujZPrzedzialu(-2, 5);
            zmianaDabli = losujZPrzedzialu(-3, 5);
            zmianaPsychiki = losujZPrzedzialu(-20, 10);
            dodaneMecze = losujZPrzedzialu(15, 30);
            dodanyExp = 50;
            fazaTurnieju = etapy[losujZPrzedzialu(1, 4)];
        }
        else if (liga === "bielska") {
            zmianaSredniej = losujZPrzedzialu(-5, 8);
            zmianaDabli = losujZPrzedzialu(-5, 7);
            zmianaPsychiki = losujZPrzedzialu(-35, 15);
            dodaneMecze = losujZPrzedzialu(10, 25);
            dodanyExp = 100;
            fazaTurnieju = etapy[losujZPrzedzialu(0, 6)];
        }

        gracz.srednia += zmianaSredniej;
        gracz.dable += zmianaDabli;
        gracz.psychika += zmianaPsychiki;
        gracz.mecze += dodaneMecze;
        gracz.exp += dodanyExp;

        // Limity
        if (gracz.psychika > 100) gracz.psychika = 100;
        if (gracz.psychika < 0) gracz.psychika = 0;
        if (gracz.dable > 100) gracz.dable = 100;
        if (gracz.dable < 0) gracz.dable = 0;
        if (gracz.srednia < 10) gracz.srednia = 10;

        generujPodsumowanie(zmianaSredniej, zmianaPsychiki, dodaneMecze, fazaTurnieju);
    }

    function generujPodsumowanie(zmSredniej, zmPsychiki, m, faza) {
        document.getElementById('ekran-rozgrywek').style.display = 'none';
        document.getElementById('ekran-podsumowania').style.display = 'block';

        let psychikaKolor = zmPsychiki < 0 ? "#ff4444" : "#4caf50";
        let sredniaKolor = zmSredniej < 0 ? "#ff4444" : "#4caf50";

        let raport = `
            <div class="log-entry"><b>Mamy Grudzień. Zbliżają się Mistrzostwa Świata, ale jako amator na razie oglądasz je w telewizji.</b></div>
            <hr>
            <div class="log-entry">Rozegrane mecze w tym sezonie: ${m} (Łącznie: ${gracz.mecze})</div>
            <div class="log-entry">Najlepszy wynik w turnieju: <b>${faza}</b></div>
            <br>
            <div class="log-entry">Zmiana średniej: <span style="color:${sredniaKolor}">${zmSredniej > 0 ? '+'+zmSredniej : zmSredniej}</span> (Obecnie: ${gracz.srednia.toFixed(1)})</div>
            <div class="log-entry">Aktualna skuteczność na podwójnych: ${gracz.dable}%</div>
            <div class="log-entry">Wpływ na psychikę: <span style="color:${psychikaKolor}">${zmPsychiki > 0 ? '+'+zmPsychiki : zmPsychiki}</span> (Stan: ${gracz.psychika}/100)</div>
            <div class="log-entry">Zdobyte doświadczenie: ${gracz.exp} punktów</div>
        `;

        if (gracz.psychika < 20) {
            raport += `<br><div class="log-entry" style="color:#ff4444"><b>OSTRZEŻENIE:</b> Twoja psychika leży i kwiczy. Jesteś na skraju dartitis. Odpocznij.</div>`;
        }

        document.getElementById('log').innerHTML = raport;
    }

    function nastepnySezon() {
        przejdzDoRozgrywek();
    }
</script>
