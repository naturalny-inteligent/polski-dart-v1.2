function symulujSezon() { // 1. Sprawdzamy co zaznaczyłeś w radiu let wybory = document.getElementsByName('liga'); let wybranaLiga = "boty"; for (let i = 0; i < wybory.length; i++) { if (wybory[i].checked) wybranaLiga = wybory[i].value; } // 2. Mapowanie wyboru na TIER z bazy let docelowyTier = 5; if (wybranaLiga === "poznan") docelowyTier = 4; else if (wybranaLiga === "bielska") docelowyTier = 3; // Pobieramy turnieje tylko z wybranego tieru let dostepneTurnieje = bazaTurniejow.filter(t => t.tier === docelowyTier); if (dostepneTurnieje.length === 0) dostepneTurnieje = bazaTurniejow; // Zabezpieczenie let liczbaTurniejowWSezonie = losujZPrzedzialu(10, 15); let statystykiSezonu = {}; let najwyzszySukces = "Brak sukcesów. Dno i wodorosty."; let maxIndeksEtapu = -1; let rozegraneMecze = 0; // 3. Symulacja każdego turnieju w sezonie for (let i = 0; i < liczbaTurniejowWSezonie; i++) { // Losujemy konkretny turniej z puli dostępnych dla danego tieru let turniej = dostepneTurnieje[losujZPrzedzialu(0, dostepneTurnieje.length - 1)]; let fazaOstatnia = "Odpadnięcie przed turniejem"; let lokalnyMaxIndeks = -1; // Symulacja przejścia przez drabinkę for (let j = 0; j < turniej.etapy.length; j++) { let etap = turniej.etapy[j]; rozegraneMecze++; // Rzut kością: Twoja średnia +- odchylenie vs wymagana średnia let rzut = losujZPrzedzialu(-5, 7); let mojaFormaWMeczu = gracz.srednia + rzut; if (mojaFormaWMeczu >= etap.sr) { fazaOstatnia = etap.faza; lokalnyMaxIndeks = j; // Jeśli to finał i wygraliśmy: if (j === turniej.etapy.length - 1) { fazaOstatnia = "ZWYCIĘSTWO W TURNIEJU"; } } else { break; // Odpadasz, pętla drabinki się kończy } } // Zapisujemy, na jakim etapie gracz zakończył ten konkretny turniej if (statystykiSezonu[fazaOstatnia]) { statystykiSezonu[fazaOstatnia]++; } else { statystykiSezonu[fazaOstatnia] = 1; } // Aktualizacja największego sukcesu w całym sezonie if (lokalnyMaxIndeks > maxIndeksEtapu) { maxIndeksEtapu = lokalnyMaxIndeks; najwyzszySukces = `${fazaOstatnia} (${turniej.nazwa})`; } } // 4. Kalkulacja postępów po sezonie let zmianaSredniej = losujZPrzedzialu(-2, 3); if (maxIndeksEtapu >= 2) zmianaSredniej += losujZPrzedzialu(1, 3); // Bonus za głębokie zajście w drabince if (maxIndeksEtapu === -1) zmianaSredniej -= losujZPrzedzialu(1, 2); // Kara za bycie beznadziejnym let zmianaPsychiki = losujZPrzedzialu(-15, 10); if (maxIndeksEtapu >= 3) zmianaPsychiki += 20; gracz.srednia += zmianaSredniej; gracz.psychika += zmianaPsychiki; gracz.mecze += rozegraneMecze; gracz.exp += rozegraneMecze * 10; // Blokady limitów if (gracz.psychika > 100) gracz.psychika = 100; if (gracz.psychika < 0) gracz.psychika = 0; gracz.dable = Math.max(5, Math.floor(gracz.srednia / 2.5) + losujZPrzedzialu(-5, 5)); generujPodsumowanie(liczbaTurniejowWSezonie, statystykiSezonu, najwyzszySukces, zmianaSredniej, zmianaPsychiki, rozegraneMecze); } function generujPodsumowanie(ileTurniejow, statystyki, maxSukces, zmSredniej, zmPsychiki, m) { document.getElementById('ekran-rozgrywek').style.display = 'none'; document.getElementById('ekran-podsumowania').style.display = 'block'; let psychikaKolor = zmPsychiki < 0 ? "#ff4444" : "#4caf50"; let sredniaKolor = zmSredniej < 0 ? "#ff4444" : "#4caf50"; let raport = `
KONIEC SEZONU. Wystąpiłeś w ${ileTurniejow} turniejach.
Największy sukces w tym roku: ${maxSukces}

Historia startów:
`; // Generowanie listy wyników (ile razy gdzie odpadłeś) for (let faza in statystyki) { raport += `
- Zakończono na: ${faza} (Ilość: ${statystyki[faza]})
`; } raport += `
Rozegrane mecze łącznie w sezonie: ${m}
Zmiana średniej: ${zmSredniej > 0 ? '+'+zmSredniej : zmSredniej} (Nowa bazowa: ${gracz.srednia.toFixed(1)})
Aktualna skuteczność na podwójnych: ${gracz.dable}%
Wpływ na psychikę: ${zmPsychiki > 0 ? '+'+zmPsychiki : zmPsychiki} (Stan: ${gracz.psychika}/100)
Całkowite doświadczenie: ${gracz.exp} punktów
`; if (gracz.psychika < 20) { raport += `
OSTRZEŻENIE: Psychika w gruzach. Jeśli nie odpoczniesz, złapiesz dartitis.
